cmake_minimum_required(VERSION 3.24)
project(COD2Coop VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "COD2 CO-OP requires Windows x86. Configure with -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc /utf-8)
endif()
add_library(coop_platform STATIC src/compatibility/platform.cpp)
target_include_directories(coop_platform PUBLIC src)
target_link_libraries(coop_platform PUBLIC bcrypt advapi32)
add_library(coop_network STATIC src/networking/protocol.cpp src/networking/session.cpp)
target_include_directories(coop_network PUBLIC src)
target_link_libraries(coop_network PUBLIC ws2_32)
add_library(COD2CoopRuntime SHARED src/bootstrap/runtime.cpp src/engine/bridge.cpp src/engine/experimental_player2.cpp src/engine/player2_presentation.cpp src/replication/native_world.cpp)
target_link_libraries(COD2CoopRuntime PRIVATE coop_platform coop_network user32)
add_executable(COD2CoopLauncher launcher/main.cpp)
target_link_libraries(COD2CoopLauncher PRIVATE coop_platform user32 advapi32)
add_executable(COD2CoopControl WIN32 launcher/gui.cpp)
target_link_libraries(COD2CoopControl PRIVATE iphlpapi ws2_32 comctl32 shell32 user32 gdi32)
add_executable(COD2CoopTests tests/unit/core_tests.cpp)
target_link_libraries(COD2CoopTests PRIVATE coop_network)
add_executable(COD2CoopEndpoint tests/network/endpoint.cpp)
target_link_libraries(COD2CoopEndpoint PRIVATE coop_network)
add_executable(COD2CoopRuntimePeer tests/network/runtime_peer.cpp)
target_link_libraries(COD2CoopRuntimePeer PRIVATE coop_network)
add_executable(COD2CoopWorldTests tests/unit/native_world_tests.cpp src/replication/native_world.cpp)
target_link_libraries(COD2CoopWorldTests PRIVATE coop_network)
enable_testing()
add_test(NAME protocol_session_authority COMMAND COD2CoopTests)
add_test(NAME native_world_tracking COMMAND COD2CoopWorldTests)
add_test(NAME native_launcher_ui COMMAND COD2CoopControl --smoke-test)
install(TARGETS COD2CoopRuntime COD2CoopLauncher COD2CoopControl COD2CoopEndpoint RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES tools/Stage-NativeUi.ps1 DESTINATION tools)
install(DIRECTORY assets/original-mod-assets-only/ui DESTINATION assets/original-mod-assets-only)
