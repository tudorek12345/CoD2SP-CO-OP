cmake_minimum_required(VERSION 3.24)
project(COD2CoopDiagnostics LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with -A Win32; target game is x86.")
endif()
add_executable(COD2CoopProcessStacks process_stacks.cpp)
target_compile_features(COD2CoopProcessStacks PRIVATE cxx_std_17)
target_compile_definitions(COD2CoopProcessStacks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(COD2CoopProcessStacks PRIVATE dbghelp)
